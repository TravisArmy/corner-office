const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type" };
const json = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...cors, "Content-Type": "application/json" } });

function easternToday(): string {
  const p: Record<string, string> = {};
  new Intl.DateTimeFormat("en-US", { timeZone: "America/New_York", year: "numeric", month: "2-digit", day: "2-digit" }).formatToParts(new Date()).forEach((x) => (p[x.type] = x.value));
  return `${p.year}-${p.month}-${p.day}`;
}
function lastWeekdays(n: number): string[] {
  const out: string[] = [], [y, m, d] = easternToday().split("-").map(Number), day = new Date(Date.UTC(y, m - 1, d));
  while (out.length < n) { const g = day.getUTCDay(); if (g !== 0 && g !== 6) out.unshift(day.toISOString().slice(0, 10)); day.setUTCDate(day.getUTCDate() - 1); }
  return out;
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  const key = Deno.env.get("MASSIVE_API_KEY");
  if (!key) return json({ error: "MASSIVE_API_KEY is not set in Edge Function secrets." }, 500);
  const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!);

  // at most one fetch every 10 minutes, however many players ask
  const { data: meta } = await sb.from("market_meta").select("last_try").eq("id", 1).maybeSingle();
  if (meta && Date.now() - new Date(meta.last_try).getTime() < 10 * 60 * 1000) return json({ ok: true, skipped: "checked recently" });
  await sb.from("market_meta").upsert({ id: 1, last_try: new Date().toISOString() });

  const { data: watch } = await sb.from("watch").select("t").limit(1000);
  const want = new Set<string>([...GAME_TICKERS, ...((watch ?? []).map((w: { t: string }) => w.t))]);

  // the last four weekdays are re-fetched together so prices stay consistent after a stock split
  const report: Record<string, unknown> = {};
  for (const d of lastWeekdays(4)) {
    const r = await fetch(`https://api.massive.com/v2/aggs/grouped/locale/us/market/stocks/${d}?adjusted=true&include_otc=true&apiKey=${key}`);
    if (!r.ok) { report[d] = `no data yet (${r.status})`; continue; }
    const body = await r.json();
    const rows = (body.results ?? []).filter((x: { T: string; c: number }) => want.has(x.T) && x.c > 0).map((x: { T: string; c: number }) => ({ d, t: x.T, c: x.c }));
    if (rows.length < 100) { report[d] = "market closed or not published yet"; continue; }
    for (let i = 0; i < rows.length; i += 500) { const { error } = await sb.from("closes").upsert(rows.slice(i, i + 500)); if (error) return json({ error: error.message }, 500); }
    await sb.from("market_days").upsert({ d, n: rows.length, fetched_at: new Date().toISOString() });
    report[d] = rows.length + " prices saved";
  }
  return json({ ok: true, days: report });
});
